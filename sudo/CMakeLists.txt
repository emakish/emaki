cmake_minimum_required(VERSION 3.25)
project(emaki_sudo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 REQUIRED COMPONENTS Core Qml)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd>=256)
pkg_get_variable(SYSTEMD_UTIL_DIR systemd systemdutildir)
find_library(PAM_LIBRARY pam REQUIRED)
find_program(SUDO_EXECUTABLE sudo REQUIRED)
find_program(QUICKSHELL_EXECUTABLE NAMES qs quickshell REQUIRED)
find_program(NIRI_EXECUTABLE NAMES niri-emaki REQUIRED)
find_program(SYSTEMD_EXECUTABLE NAMES systemd HINTS "${SYSTEMD_UTIL_DIR}" REQUIRED)
file(REAL_PATH "${SUDO_EXECUTABLE}" SUDO_EXECUTABLE_REAL)
file(REAL_PATH "${QUICKSHELL_EXECUTABLE}" QUICKSHELL_EXECUTABLE_REAL)
file(REAL_PATH "${NIRI_EXECUTABLE}" NIRI_EXECUTABLE_REAL)
file(REAL_PATH "${SYSTEMD_EXECUTABLE}" SYSTEMD_EXECUTABLE_REAL)
set(EMAKI_DATADIR "${CMAKE_INSTALL_PREFIX}/share/emaki" CACHE PATH "Emaki data directory")
set(EMAKI_LIBEXECDIR "${CMAKE_INSTALL_PREFIX}/libexec/emaki" CACHE PATH "Emaki helpers")
set(EMAKI_PAM_MODULE "${CMAKE_INSTALL_PREFIX}/lib/security/pam_emaki_sudo.so" CACHE FILEPATH "Sudo PAM module")
add_library(sudo-core STATIC core.cpp timestamps.cpp cleanup.cpp warning.cpp)
target_compile_options(sudo-core PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(sudo-core PUBLIC
    EMAKI_SUDO_PATH="${SUDO_EXECUTABLE_REAL}"
    EMAKI_PAM_MODULE="${EMAKI_PAM_MODULE}"
    EMAKI_QUICKSHELL_PATH="${QUICKSHELL_EXECUTABLE_REAL}"
    EMAKI_NIRI_PATH="${NIRI_EXECUTABLE_REAL}"
    EMAKI_SYSTEMD_PATH="${SYSTEMD_EXECUTABLE_REAL}"
    EMAKI_COMMAND_BINDIR="${CMAKE_INSTALL_PREFIX}/bin"
    EMAKI_SHELL_PATH="${EMAKI_DATADIR}/shell"
    EMAKI_SUDO_KEY_MARKER="${EMAKI_DATADIR}/sudo-key.installed")
foreach(name broker cli)
    add_executable(emaki-sudo-${name} ${name}.cpp)
    target_link_libraries(emaki-sudo-${name} PRIVATE sudo-core)
    target_compile_options(emaki-sudo-${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
target_link_libraries(emaki-sudo-broker PRIVATE PkgConfig::SYSTEMD)
add_library(pam_emaki_sudo MODULE pam.cpp)
set_target_properties(pam_emaki_sudo PROPERTIES PREFIX "")
target_link_libraries(pam_emaki_sudo PRIVATE sudo-core ${PAM_LIBRARY})
target_compile_options(pam_emaki_sudo PRIVATE -Wall -Wextra -Wpedantic -Werror)
configure_file(pam-service.in emaki-sudo-pam @ONLY)
configure_file(emaki-sudo.service.in emaki-sudo.service @ONLY)
configure_file(emaki-sudo-cleanup.service.in emaki-sudo-cleanup.service @ONLY)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/install-paths"
    CONTENT "${CMAKE_INSTALL_PREFIX}|${EMAKI_DATADIR}|${EMAKI_LIBEXECDIR}\n")
if(TARGET emaki-sudo-cli)
    set_target_properties(emaki-sudo-cli PROPERTIES OUTPUT_NAME emaki-sudo)
endif()
add_library(emakisudoplugin MODULE ui.cpp ui.hpp)
set_target_properties(emakisudoplugin PROPERTIES AUTOMOC ON
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/qml/Emaki/Sudo")
target_link_libraries(emakisudoplugin PRIVATE sudo-core Qt6::Core Qt6::Qml)
target_compile_options(emakisudoplugin PRIVATE -Wall -Wextra -Wpedantic -Werror)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/qml/Emaki/Sudo")
configure_file(qmldir "${CMAKE_CURRENT_BINARY_DIR}/qml/Emaki/Sudo/qmldir" COPYONLY)
configure_file(plugins.qmltypes "${CMAKE_CURRENT_BINARY_DIR}/qml/Emaki/Sudo/plugins.qmltypes" COPYONLY)
enable_testing()
add_executable(sudo-client-tests client-tests.cpp ui.cpp ui.hpp)
set_target_properties(sudo-client-tests PROPERTIES AUTOMOC ON)
target_link_libraries(sudo-client-tests PRIVATE sudo-core Qt6::Core Qt6::Qml)
target_compile_options(sudo-client-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_options(sudo-client-tests PRIVATE
    -Wl,--wrap=_ZN5emaki12sudo_channel13connectBrokerEv
    -Wl,--wrap=_ZN5emaki12sudo_channel10sendPacketEiNS0_4TypeEmjPKvm
    -Wl,--wrap=_ZN5emaki12sudo_channel13receivePacketEiRNS0_6PacketE)
add_test(NAME sudo-client COMMAND sudo-client-tests)
add_executable(sudo-mapping-tests mapping-tests.cpp)
target_link_libraries(sudo-mapping-tests PRIVATE sudo-core PkgConfig::SYSTEMD)
target_compile_options(sudo-mapping-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME sudo-mapping COMMAND sudo-mapping-tests)
add_executable(sudo-core-tests tests.cpp)
target_link_libraries(sudo-core-tests PRIVATE sudo-core)
target_compile_options(sudo-core-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME sudo-core COMMAND sudo-core-tests)
add_library(pam_emaki_sudo_test MODULE pam.cpp pam-tests.cpp)
set_target_properties(pam_emaki_sudo_test PROPERTIES PREFIX "")
target_compile_definitions(pam_emaki_sudo_test PRIVATE EMAKI_PAM_TEST)
target_link_libraries(pam_emaki_sudo_test PRIVATE sudo-core ${PAM_LIBRARY})
target_compile_options(pam_emaki_sudo_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(sudo-pam-tests pam-control-tests.cpp)
target_link_libraries(sudo-pam-tests PRIVATE ${PAM_LIBRARY} ${CMAKE_DL_LIBS})
target_compile_options(sudo-pam-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME sudo-pam COMMAND sudo-pam-tests $<TARGET_FILE:pam_emaki_sudo_test>
    "${CMAKE_CURRENT_BINARY_DIR}/emaki-sudo-pam" "${EMAKI_PAM_MODULE}")
add_executable(sudo-arguments-tests arguments-tests.cpp)
target_compile_options(sudo-arguments-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME sudo-arguments COMMAND sudo-arguments-tests)
add_executable(sudo-pam-production-tests pam-production-tests.cpp)
target_link_libraries(sudo-pam-production-tests PRIVATE ${PAM_LIBRARY} ${CMAKE_DL_LIBS})
target_compile_options(sudo-pam-production-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_executable(sudo-lifecycle-tests lifecycle-tests.cpp)
target_link_libraries(sudo-lifecycle-tests PRIVATE sudo-core)
target_compile_options(sudo-lifecycle-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME sudo-lifecycle COMMAND sudo-lifecycle-tests)
add_executable(sudo-controller-tests controller-tests.cpp)
target_link_libraries(sudo-controller-tests PRIVATE sudo-core PkgConfig::SYSTEMD)
target_compile_options(sudo-controller-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME sudo-controller COMMAND sudo-controller-tests)
add_executable(sudo-filesystem-tests filesystem-tests.cpp)
target_link_libraries(sudo-filesystem-tests PRIVATE sudo-core PkgConfig::SYSTEMD)
target_compile_options(sudo-filesystem-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME sudo-filesystem COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/filesystem-tests.sh"
    $<TARGET_FILE:sudo-filesystem-tests>)
set_tests_properties(sudo-filesystem PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 10)
add_executable(sudo-timestamp-tests timestamp-tests.cpp)
target_link_libraries(sudo-timestamp-tests PRIVATE sudo-core)
target_compile_options(sudo-timestamp-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_options(sudo-timestamp-tests PRIVATE -Wl,--wrap=fstat,--wrap=fstatat)
add_test(NAME sudo-timestamp COMMAND sudo-timestamp-tests)
set_tests_properties(sudo-timestamp PROPERTIES TIMEOUT 10)
add_executable(sudo-cleanup-tests cleanup-tests.cpp)
target_link_libraries(sudo-cleanup-tests PRIVATE sudo-core)
target_compile_options(sudo-cleanup-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_options(sudo-cleanup-tests PRIVATE -Wl,--wrap=fstat)
add_test(NAME sudo-cleanup COMMAND sudo-cleanup-tests)
set_tests_properties(sudo-cleanup PROPERTIES TIMEOUT 20)

add_executable(sudo-warning-tests warning-tests.cpp)
target_link_libraries(sudo-warning-tests PRIVATE sudo-core)
target_compile_options(sudo-warning-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_options(sudo-warning-tests PRIVATE -Wl,--wrap=fstat)
add_test(NAME sudo-warning COMMAND sudo-warning-tests)

add_executable(sudo-fdopendir-tests fdopendir-tests.cpp)
target_link_libraries(sudo-fdopendir-tests PRIVATE sudo-core)
target_compile_options(sudo-fdopendir-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_options(sudo-fdopendir-tests PRIVATE -Wl,--wrap=fdopendir,--wrap=fstat)
add_test(NAME sudo-fdopendir-authentication COMMAND sudo-fdopendir-tests authentication)
add_test(NAME sudo-fdopendir-cleanup COMMAND sudo-fdopendir-tests cleanup)
